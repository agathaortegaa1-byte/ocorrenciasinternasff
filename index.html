Sincronização Multi-Terminal via Google Sheets API

Conectar Planilha Google (Multi-Computador)

Sincronização entre terminais

Cole a URL do Web App do Google Apps Script abaixo. Todos os computadores da operação compartilharão automaticamente os registros e os comentários em tempo real.

Passo a Passo para configurar a Planilha Google:

  1. Acesse sheets.new para criar uma nova Planilha Google.
  2. No menu superior, clique em Extensões > Apps Script.
  3. Apague o código existente e cole o script atualizado abaixo (já inclui suporte a comentários):
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var result = [];
  
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] !== "") {
      var comentariosParsed = [];
      try {
        if (data[i][8]) comentariosParsed = JSON.parse(data[i][8]);
      } catch (err) {
        comentariosParsed = [];
      }
      result.push({
        id: data[i][0],
        dataHora: data[i][1],
        colaborador: data[i][2],
        setor: data[i][3],
        tipo: data[i][4],
        penalidade: data[i][5],
        gestor: data[i][6],
        observacoes: data[i][7],
        comentarios: comentariosParsed
      });
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "Data/Hora", "Colaborador", "Setor", "Tipo", "Penalidade", "Gestor", "Observações", "ComentariosJSON"]);
  }
  
  var params = JSON.parse(e.postData.contents);
  
  if (params.action === "addComment") {
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(params.id)) {
        var existing = [];
        try {
          if (data[i][8]) existing = JSON.parse(data[i][8]);
        } catch (err) {
          existing = [];
        }
        existing.push(params.comment);
        sheet.getRange(i + 1, 9).setValue(JSON.stringify(existing));
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status: "comment_added"}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  sheet.appendRow([
    params.id,
    params.dataHora,
    params.colaborador,
    params.setor,
    params.tipo,
    params.penalidade,
    params.gestor,
    params.observacoes,
    JSON.stringify(params.comentarios || [])
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
